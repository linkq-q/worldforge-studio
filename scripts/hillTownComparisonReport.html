AI 两组都由 GPT 写首版 Scene Code，提供相同的地形观察、摆放、地基、地表与通路 API，允许自定义辅助函数；第一组按用户确认保留纯程序化基线，不调用 GPT 规划。

唯一的规划差别

  • 程序化基线：候选地块评分、站点选择与路径连接由固定代码完成，四次只改数值种子。
  • 自由 AI：只描述聚落需求，不给空间组织原则。
  • 原则引导 AI：额外提示从居民停留和往来需求判断建筑、地面与通路的关系，具体构造由 AI 决定。

function plan(api) {
  let state = 7950;
  function rand() { state = (Math.imul(state, 1664525) + 1013904223) >>> 0; return state / 4294967296; }
  function dist(a,b) { return Math.hypot(a[0]-b[0],a[1]-b[1]); }
  const sites=[];
  for (let x=-34; x<=34; x+=5) for (let z=-34; z<=34; z+=5) {
    const p=[x+(rand()-.5)*2,z+(rand()-.5)*2];
    const s=api.environmentSample(p);
    const corners=[[p[0]-3,p[1]-3],[p[0]+3,p[1]-3],[p[0]-3,p[1]+3],[p[0]+3,p[1]+3]];
    const heights=corners.map(q=>api.environmentSample(q).height);
    const relief=Math.max(...heights)-Math.min(...heights);
    sites.push({p, h:s.height, relief, slope:s.slope, jitter:rand()});
  }
  sites.sort((a,b)=>(a.relief*2+Math.hypot(...a.p)*.04+a.jitter*1.2)-(b.relief*2+Math.hypot(...b.p)*.04+b.jitter*1.2));
  const center=sites[0];
  const homes=[];
  for (let i=0;i<5;i++) {
    const options=sites.filter(s=>dist(s.p,center.p)>13 && dist(s.p,center.p)<36 && homes.every(h=>dist(s.p,h.p)>12));
    options.sort((a,b)=>(a.relief*1.8+Math.abs(dist(a.p,center.p)-24)*.07+a.jitter*1.4)-(b.relief*1.8+Math.abs(dist(b.p,center.p)-24)*.07+b.jitter*1.4));
    if (options.length) homes.push(options[0]);
  }
  const nodes=[center,...homes];
  const connected=[center];
  const edges=[];
  while (connected.length<nodes.length) {
    let best=null;
    for (const a of connected) for (const b of nodes) if (!connected.includes(b)) {
      const cost=dist(a.p,b.p)+Math.abs(a.h-b.h)*2.5;
      if (!best || cost<best.cost) best={a,b,cost};
    }
    if (!best) break;
    edges.push(best); connected.push(best.b);
  }
  for (const e of edges) api.rampTerrain({start:e.a.p,end:e.b.p,width:2.6,softness:.55,strength:.9});
  for (const s of nodes) api.sculptTerrain({mode:'flatten',point:s.p,radius:s===center?6:4.8,strength:.9,targetHeight:s.h});
  for (let i=0;i<edges.length;i++) api.route({id:'footpath-'+i,points:[edges[i].a.p,edges[i].b.p],width:2.4,surface:'soil'});
  api.surface({id:'shared-ground',surface:'paving',region:{kind:'circle',x:center.p[0],z:center.p[1],radius:6.5}});
  const hall=api.place({assetId:'asset-5134f8d1-f4c8-4624',name:'村民公共屋',position:center.p,role:'structure'});
  api.foundation({under:[hall],top:'level',margin:.5,maxThickness:4});
  for (let i=0;i<homes.length;i++) {
    const s=homes[i];
    const home=api.place({assetId:'asset-c573ccb8-04a5-4b34',name:'住宅 '+(i+1),position:s.p,facing:{target:center.p},role:'structure'});
    api.foundation({under:[home],top:'level',margin:.45,maxThickness:4});
  }
  for (const s of sites) {
    if (rand()>.05 || nodes.some(n=>dist(s.p,n.p)<9) || edges.some(e=>{
      const ax=e.a.p[0],az=e.a.p[1],bx=e.b.p[0],bz=e.b.p[1];
      const t=Math.max(0,Math.min(1,((s.p[0]-ax)*(bx-ax)+(s.p[1]-az)*(bz-az))/((bx-ax)**2+(bz-az)**2)));
      return Math.hypot(s.p[0]-(ax+t*(bx-ax)),s.p[1]-(az+t*(bz-az)))<5;
    })) continue;
    api.place({assetId:'asset-c1c2f623-c0e9-4520',name:'山地树',position:s.p,role:'environment'});
  }

  function inspect(x, z, angle, communal) {
    var sx = Math.sin(angle), cz = Math.cos(angle);
    var hw = communal ? 5 : 4.02;
    var hd = communal ? 4.32 : 3.66;
    var low = Infinity, high = -Infinity;
    var samples = [[0, 0], [-hw, -hd], [hw, -hd], [-hw, hd], [hw, hd], [0, hd]];
    for (var j = 0; j < samples.length; j++) {
      var u = samples[j][0], v = samples[j][1];
      var h = height(x + u * cz + v * sx, z - u * sx + v * cz);
      low = Math.min(low, h);
      high = Math.max(high, h);
    }
    var door = [x + sx * (hd + 0.35), z + cz * (hd + 0.35)];
    var approach = [x + sx * (hd + 6), z + cz * (hd + 6)];
    var entryRise = high - height(door[0], door[1]);
    var climb = high - height(approach[0], approach[1]);
    var env = api.environmentSample([x, z]);
    var wet = typeof env.waterDistance === "number" && env.waterDistance < 5 ? 60 : 0;
    return {
      x: x, z: z, angle: angle, communal: communal,
      floor: high + 0.08, door: door, approach: approach,
      score: (high - low) * (high - low) * 3 + entryRise * entryRise +
        climb * climb * 0.7 + wet
    };
  }

  for (var ix = -2; ix <= 2; ix++) {
    for (var iz = -2; iz <= 2; iz++) {

        for (var i = 0; i < 6; i++) {

          var r = i === 0 ? 19.5 : 18;
          var x = cx + Math.sin(a) * r;
          var z = zz + Math.cos(a) * r;
          var b = inspect(x, z, a + Math.PI, i === 0);
          score += b.score + 0.12 * Math.pow(b.floor - centerH, 2);
          buildings.push(b);
        }
        var cmin = centerH, cmax = centerH;
        for (var q = 0; q < 6; q++) {

          cmin = Math.min(cmin, ch);
          cmax = Math.max(cmax, ch);

      }
    }
  }

    id: "village-common",
    surface: "soil",
    region: { kind: "circle", x: best.x, z: best.z, radius: 5.8 },
    intensity: 0.85
  });
  api.surface({
    id: "common-stone-center",
    surface: "rock",
    region: { kind: "circle", x: best.x, z: best.z, radius: 2.5 },

    });
    api.foundation({
      under: [ref],
      shape: "rounded-rectangle",
      top: "level",

      thickness: 0.2,

    var ux = Math.sin(ea), uz = Math.cos(ea);
    var p0 = [best.x + ux * 7, best.z + uz * 7];
    var p1 = [best.x + ux * 28, best.z + uz * 28];
    var p2 = [Math.max(-44, Math.min(44, best.x + ux * 40)),

    var t = den ? ((x - a[0]) * dx + (z - a[1]) * dz) / den : 0;
    t = Math.max(0, Math.min(1, t));
    return Math.sqrt(Math.pow(x - a[0] - t * dx, 2) + Math.pow(z - a[1] - t * dz, 2));

  var house = "asset-c573ccb8-04a5-4b34";
  var hall = "asset-5134f8d1-f4c8-4624";
  var tree = "asset-c1c2f623-c0e9-4520";
  var tau = Math.PI * 2;

  function height(x, z) {
    return api.environmentSample([x, z]).height;
  }
  function polar(cx, cz, a, r) {
    return [cx + Math.sin(a) * r, cz + Math.cos(a) * r];
  }
  function distance(a, b) {
    return Math.hypot(a[0] - b[0], a[1] - b[1]);
  }
  function footprint(x, z, a, w, d) {
    var result = [height(x, z)];
    for (var i = 0; i < 4; i++) {
      var u = (i % 2 === 0 ? -1 : 1) * w;
      var v = (i < 2 ? -1 : 1) * d;
      result.push(height(
        x + Math.cos(a) * u + Math.sin(a) * v,
        z - Math.sin(a) * u + Math.cos(a) * v
      ));
    }
    return result;
  }

  var best = null;
  for (var ix = -2; ix <= 2; ix++) {
    for (var iz = -2; iz <= 2; iz++) {
      for (var turn = 0; turn < 8; turn++) {
        var cx = ix * 5;
        var cz = iz * 5;
        var angle = turn * tau / 8;
        var centerH = height(cx, cz);
        var score = 0;
        var buildings = [];
        for (var n = 0; n < 6; n++) {
          var a = angle + n * tau / 6;
          var radius = n === 0 ? 21 : 23;
          var p = polar(cx, cz, a, radius);
          var samples = footprint(p[0], p[1], a, n === 0 ? 5.4 : 4.4, n === 0 ? 4.7 : 4.1);
          var high = Math.max.apply(null, samples);
          var low = Math.min.apply(null, samples);
          var mean = 0;
          for (var k = 0; k < samples.length; k++) mean += samples[k] / samples.length;
          var approach = polar(cx, cz, a, 12);
          var approachH = height(approach[0], approach[1]);
          var s = api.environmentSample(p);
          score += 5 * (high - low) * (high - low);
          score += 1.4 * (high - approachH) * (high - approachH);
          score += 0.15 * (mean - centerH) * (mean - centerH);
          if (typeof s.waterDistance === "number" && s.waterDistance < 5) {
            score += 100 * (5 - s.waterDistance);
          }
          buildings.push({p: p, a: a, radius: radius, public: n === 0, top: high + 0.12});
        }
        var plazaHeights = [];
        for (var j = 0; j < 8; j++) {
          var q = polar(cx, cz, j * tau / 8, 4.2);
          plazaHeights.push(height(q[0], q[1]));
        }
        var plazaRange = Math.max.apply(null, plazaHeights) - Math.min.apply(null, plazaHeights);
        score += 12 * plazaRange * plazaRange;
        var previous = null;
        for (var j = 0; j <= 24; j++) {
          var q = polar(cx, cz, angle + j * tau / 24, 12);
          var h = height(q[0], q[1]);
          if (previous !== null) score += 3 * (h - previous) * (h - previous);
          previous = h;
        }
        if (!best || score < best.score) {
          best = {score: score, cx: cx, cz: cz, angle: angle, buildings: buildings, plazaH: centerH};

      }

    surface: "paving",

  });

    var p = polar(cx, cz, best.angle + i * tau / 24, 12);

  }

    for (var j = 0; j < points.length - 1; j++) {
      api.rampTerrain({
        start: points[j], end: points[j + 1], width: width + 0.5,
        startHeight: heights[j], endHeight: heights[j + 1],
        softness: 0.65, strength: 1
      });
      paths.push([points[j], points[j + 1], width]);
    }
    api.route({id: id, points: points, width: width, surface: surface});
  }

  var closedRing = ring.slice();
  var closedHeights = ringH.slice();
  closedRing.push(ring[0]);
  closedHeights.push(ringH[0]);
  gradedPath("village-contour-lane", closedRing, closedHeights, 2.5, "soil");

  for (var n = 0; n < 6; n++) {

    var landing = polar(cx, cz, b.a, b.radius - depth - 1.35);
    var startIndex = n * 4;
    var start = ring[startIndex];
    var startH = ringH[startIndex];
    var delta = b.top - startH;
    var points = [start];
    var heights = [startH];

    if (Math.abs(delta) / distance(start, landing) > 0.23) {
      var side = ringH[(startIndex + 1) % 24] < ringH[(startIndex + 23) % 24] ? 1 : -1;
      var alternateIndex = (startIndex + side * 2 + 24) % 24;
      start = ring[alternateIndex];
      startH = ringH[alternateIndex];
      points = [start];
      heights = [startH];
      var elbow = polar(cx, cz, b.a + side * 0.32, b.radius - depth - 2.7);
      var length1 = distance(start, elbow);
      var length2 = distance(elbow, landing);
      points.push(elbow);
      heights.push(startH + (b.top - startH) * length1 / (length1 + length2));
    }
    points.push(landing);
    heights.push(b.top);
    points.push(door);
    heights.push(b.top);
    gradedPath(
      b.public ? "public-house-access" : "home-access-" + n,
      points, heights, b.public ? 3 : 1.8, b.public ? "paving" : "rock"
    );
    api.sculptTerrain({
      mode: "flatten", point: landing, radius: b.public ? 2.1 : 1.6,
      targetHeight: b.top, strength: 1
    });
    api.surface({
      id: "entrance-terrace-" + n, surface: "paving",
      region: {kind: "circle", x: landing[0], z: landing[1], radius: b.public ? 2 : 1.5},
      intensity: 1
    });
    var ref = api.place({
      assetId: b.public ? hall : house,
      name: b.public ? "居民公共屋" : "山地住宅 " + n,
      position: [b.p[0], b.top, b.p[1]],
      facing: {target: [cx, cz]},
      role: b.public ? "community" : "residence"
    });
    api.foundation({
      under: [ref], shape: "rounded-rectangle", top: "level",
      margin: 0.65, thickness: 0.2, maxThickness: 8
    });
  }

  for (var k = 0; k < 3; k++) {
    var index = k * 8;
    var a = best.angle + index * tau / 24;
    var inner = polar(cx, cz, a, 3.5);
    var mid = polar(cx, cz, a + 0.13, 7.5);
    gradedPath("court-connection-" + k, [inner, mid, ring[index]],
      [plazaH, (plazaH + ringH[index]) / 2, ringH[index]], 2.3, "rock");
  }

  var exitIndex = 0;
  var exitScore = Infinity;
  var exitEnd = null;
