function relative(key, name, anchor, dx, dz, dimensions, rotationY, relation, range, role = "functional") {
  const distance = Math.hypot(dx, dz);
  if (distance < range[0] || distance > range[1]) {
    throw new Error("Invalid relation: " + relation);
  }
  return floor(key, name, anchor.x + dx, anchor.z + dz, dimensions, rotationY, role);
}
