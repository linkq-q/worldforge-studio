  function segmentHits(r, b, pad) {
    let lo = 0, hi = 1;
    for (const [a, v, min, max] of [
      [r.ax, r.bx - r.ax, b.x - b.w / 2 - pad, b.x + b.w / 2 + pad],
      [r.az, r.bz - r.az, b.z - b.d / 2 - pad, b.z + b.d / 2 + pad]

      if (Math.abs(v) < 1e-8) {
        if (a < min || a > max) return false;
      } else {
        let t0 = (min - a) / v, t1 = (max - a) / v;
        if (t0 > t1) { const t = t0; t0 = t1; t1 = t; }
        lo = Math.max(lo, t0); hi = Math.min(hi, t1);
        if (lo > hi) return false;
      }
    }
    return true;
  }

  function clearanceKept(g) {
    let penalty = 0;
    for (const b of boxes(g)) {

      penalty += Math.max(0, Math.abs(b.z) + b.d / 2 - 5.72) * 100;

        penalty += overlap(b, { x: d.x, z: d.z, w: d.width + 1.5, d: d.width + 1.5 }, 0.15) * 100;

      nothingColliding(gs) + dominantFocalView(gs);
