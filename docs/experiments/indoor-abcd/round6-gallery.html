function floor(key, name, x, z, rotationY = 0, dimensions) {
  const s = specs[key];
  const id = api.place({
    assetId: api.asset(key),
    name,
    position: api.roomPoint(x, z, 0),
    dimensions: dimensions || s.dimensions,
    rotationY,
    role: s.role
  });
  const result = { id, x, z, rotationY };
  placed[name] = result;
  return result;
}
function related(key, name, anchor, dx, dz, range, rotationY = 0, dimensions) {
  const distance = Math.hypot(dx, dz);
  if (distance < range[0] - 0.001 || distance > range[1] + 0.001) {
    throw new Error("Relationship outside declared range: " + name);
  }
  return floor(key, name, anchor.x + dx, anchor.z + dz, rotationY, dimensions);
}

const definition = key => families.find(f => f[0] === key);
function floor(key, name, x, z, rotationY = 0, dimensions) {
  const d = definition(key);
  return api.place({
    assetId: api.asset(key),
    name,
    position: api.roomPoint(x, z, 0),
    rotationY,
    dimensions: dimensions || d[2],
    role: d[3]
  });
}
function supported(key, name, parentId, offset = [0, 0], dimensions) {
  const d = definition(key);
  return api.attach({
    assetId: api.asset(key),
    name,
    parentId,
    kind: "supported",
    offset,
    anchorY: "bottom",
    dimensions: dimensions || d[2],
    role: d[3]
  });
}
function overhead(name, x, z, diameter, drop) {
  const dimensions = [diameter, 0.44, diameter];
  return api.place({
    assetId: api.asset("pendant"),
    name,
    position: api.ceilingPoint(x, z, dimensions[1], drop),
    dimensions,
    role: "decor"
  });
}
function related(key, name, anchor, delta, distanceRange, rotationY = 0, dimensions) {
  const distance = Math.sqrt(delta[0] * delta[0] + delta[1] * delta[1]);
  if (distance < distanceRange[0] || distance > distanceRange[1]) {
    throw new Error("Invalid spatial relationship: " + name);
  }
  return floor(key, name, anchor.x + delta[0], anchor.z + delta[1], rotationY, dimensions);
}

// Private treatment rooms are genuinely enclosed; each front wall is declared
// whole and punched once. Inward door-swing areas remain entirely empty.
function buildTreatmentRoom(id, zone, doorX) {
  const x0 = zone.x[0], x1 = zone.x[1];
  const z0 = zone.z[0], z1 = zone.z[1];
  const wall = (suffix, from, to, openings) => api.interiorWall({
    id: id + "-" + suffix,
    from,
    to,
    thickness: 0.12,
    height: 4,
    wallType: "solid",
    color: "#E7DED0",
    ...(openings ? { openings } : {})
  });
  wall("rear", [x0, z0], [x1, z0]);
  wall("outer", [x0, z0], [x0, z1]);
  wall("inner", [x1, z0], [x1, z1]);
  wall("front", [x0, z1], [x1, z1], [{
    id: "entry",
    kind: "door",
    offset: doorX - (x0 + x1) / 2,
    bottom: 0,
    width: 1.1,
    height: 2.35
  }]);
  api.place({
    assetId: api.asset("door"),
    name: id === "treatment-a" ? "静养护理室门" : "舒缓护理室门",
    roomOpeningId: id + "-front/entry",
    dimensions: [1.1, 2.35, 0.10],
    role: "functional"
  });
}
